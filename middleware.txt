// middleware.ts (letakkan di root atau src/)
import { createServerClient, type CookieOptions } from '@supabase/ssr'
import { NextResponse, type NextRequest } from 'next/server'

export async function middleware(request: NextRequest) {
    console.log(`\n--- MIDDLEWARE START --- Path: ${request.nextUrl.pathname}`); // Log awal middleware
    let response = NextResponse.next({ request: { headers: request.headers } });
    const incomingCookies = request.cookies.getAll();
    console.log("MIDDLEWARE: Incoming Cookies:", JSON.stringify(incomingCookies, null, 2)); // Log cookie masuk
  
    // Cari cookie auth supabase secara manual di log
    const authCookie = incomingCookies.find(cookie => cookie.name.startsWith('sb-') && cookie.name.endsWith('-auth-token'));
    if(authCookie) {
        console.log("MIDDLEWARE: Supabase auth cookie FOUND in request!");
    } else {
        console.warn("MIDDLEWARE: Supabase auth cookie NOT FOUND in request!"); // <-- Perhatikan ini!
    }
  

  // Buat instance Supabase client khusus untuk middleware
  const supabase = createServerClient(
    process.env.NEXT_PUBLIC_SUPABASE_URL!,
    process.env.NEXT_PUBLIC_SUPABASE_ANON_KEY!,
    {
      cookies: {
        get(name: string) {
          return request.cookies.get(name)?.value
        },
        set(name: string, value: string, options: CookieOptions) {
            console.log(`MIDDLEWARE: Setting cookie - ${name}`); // Log saat set
          // Jika cookie di set di middleware, perbarui request agar route handler berikutnya bisa membacanya
          request.cookies.set({ name, value, ...options })
          response = NextResponse.next({ // Buat response baru agar cookie di set
            request: {
              headers: request.headers,
            },
          })
          response.cookies.set({ name, value, ...options })
        },
        remove(name: string, options: CookieOptions) {
           // Jika cookie di remove di middleware, perbarui request dan response
          request.cookies.set({ name, value: '', ...options })
           response = NextResponse.next({
            request: {
              headers: request.headers,
            },
          })
          response.cookies.set({ name, value: '', ...options })
        },
      },
    }
  )

  // ********** PENTING: Refresh session **********
  // Ini penting untuk menjaga sesi tetap aktif dan tersedia untuk server components / actions
  // getUser() akan otomatis me-refresh token jika perlu
  await supabase.auth.getUser()
  // ***********************************************

  return response // Kembalikan response yang mungkin sudah diperbarui cookie-nya
}

// Pastikan middleware berjalan di path yang benar
export const config = {
  matcher: [
    /*
     * Cocokkan semua path request kecuali untuk:
     * - _next/static (file statis)
     * - _next/image (optimasi gambar)
     * - favicon.ico (file favicon)
     * Anda mungkin perlu menyesuaikan ini lebih lanjut tergantung struktur aplikasi Anda.
     */
    '/((?!_next/static|_next/image|favicon.ico).*)',
  ],
}
